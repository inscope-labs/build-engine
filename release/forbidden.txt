# Independent second line of defence: verify.sh fails the build if the FINISHED artifact matches any rule
# here, even when release/manifest.txt allows the path. Format:  path: <ERE>   or   text: <ERE>
# path rules match the artifact-relative file path; text rules match file contents.
#
# Text rules are matched against binary files as well (grep -a). A rule that only matched text files would be
# bypassed by putting the payload in a file with a NUL byte in it. The shipped payload contains no binary file,
# so the specific shapes below cannot match one by accident.

# -- the gateway, by location and by name
path: (^|/)gateway(/|$)
path: (^|/)ctxgate\.py$
path: (^|/)test_ctxgate\.py$
path: (^|/)gw[^/]*\.json$
# -- the gateway, by code (so a renamed copy is still caught)
# The patterns are not anchored to the start of a line: an indented copy of the gateway is still the gateway.
text: class[[:space:]]+Gateway[(:]
text: class[[:space:]]+GitSource[(:]
text: def[[:space:]]+make_server\(

# -- development-only trees that must never reach a consumer
path: (^|/)(conformance|variants)(/|$)
# -- the conformance vectors, by content (a renamed copy is still caught)
text: "vectors_version"

# -- the write gateway (gateway/write, a Cloudflare Worker), by location, config and code
path: (^|/)wrangler\.toml$
path: (^|/)\.dev\.vars
path: (^|/)\.wrangler(/|$)
path: (^|/)node_modules(/|$)
text: cloudflare:workers
text: DurableObjectState

# -- repository internals, local state, key material
path: (^|/)\.git(/|$)
path: (^|/)\.env
path: \.(pem|key|p12|pat|token)$
path: (^|/)__pycache__(/|$)
path: \.pyc$
path: (^|/)\.build-engine\.lock$
path: (^|/)\.DS_Store$
path: (^|/)[^/]*~$

# -- credential shapes. These match real tokens, not the scanner patterns in cbridge/plan_tool.py
# (those contain a bracket right after the prefix, which these require to be a token character).
text: ghp_[A-Za-z0-9]{36}
text: github_pat_[A-Za-z0-9_]{50,}
text: AKIA[0-9A-Z]{16}
text: -----BEGIN [A-Z ]*PRIVATE KEY-----
